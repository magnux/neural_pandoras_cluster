\def\nvmap#1#2#3#4#5{\begin{scope}[shift={(#2,#3)}]
\path[draw=nvBlue,line width=.68pt,fill=nvMid] ({-#4/2},{#4/2})--++(#5,{.55*#5})--++(#4,0)--++({-#5},{-.55*#5})--cycle;
\path[draw=nvBlue,line width=.68pt,fill=nvDark] ({#4/2},{-#4/2})--++(#5,{.55*#5})--++(0,#4)--++({-#5},{-.55*#5})--cycle;
\path[draw=nvBlue,line width=.68pt,fill=nvLight] ({-#4/2},{-#4/2}) rectangle ({#4/2},{#4/2});
\foreach \ii in {.25,.5,.75}{\draw[white,opacity=.43,line width=.32pt] ({#4/2+#5*\ii},{-#4/2+.55*#5*\ii+.015})--++(0,{#4-.03});}
\coordinate (#1L) at ({-#4/2},0); \coordinate (#1R) at ({#4/2+#5},{.275*#5});\coordinate (#1B) at (0,{-#4/2});\coordinate (#1T) at (0,{#4/2+.55*#5});
\end{scope}}
